  <script>
    document.addEventListener('DOMContentLoaded', () => {
      // Initialize module functions
      initNidVerification();
      initCustomRotationConfigurator();
      initDocumentVerification();
      initFormSubmission();
    });

    /**
     * 1. Smart NID Verification Component Logic
     */
    function initNidVerification() {
      const verifyBtn = document.getElementById('btn-verify-nid');
      const nidInput = document.getElementById('smart-nid-input');
      
      if (!verifyBtn || !nidInput) return;

      verifyBtn.addEventListener('click', (e) => {
        e.preventDefault();
        const nidValue = nidInput.value.trim();

        if (nidValue.length < 10) {
          alert('Please enter a valid Smart NID number.');
          return;
        }

        // Apply loading indicator state
        verifyBtn.disabled = true;
        verifyBtn.innerHTML = `Verifying...`;

        // Simulate asynchronous verification API delay
        setTimeout(() => {
          verifyBtn.style.backgroundColor = '#10b981';
          verifyBtn.innerHTML = `✓ Verified`;
          nidInput.style.borderColor = '#10b981';
          nidInput.readOnly = true;
        }, 1000);
      });
    }

    /**
     * 2. Elective Rotation Add/Remove Configurator Logic
     */
    function initCustomRotationConfigurator() {
      const addBtn = document.getElementById('btn-add-rotation');
      const departmentSelect = document.getElementById('elective-dept-select');
      const durationSelect = document.getElementById('elective-duration-select');
      const rosterBody = document.getElementById('roster-table-body');

      if (!addBtn || !departmentSelect || !durationSelect || !rosterBody) return;

      addBtn.addEventListener('click', (e) => {
        e.preventDefault();

        const selectedDept = departmentSelect.value;
        const selectedWeeks = parseInt(durationSelect.value, 10);

        if (!selectedDept || isNaN(selectedWeeks)) {
          alert('Please select an elective department.');
          return;
        }

        // Construct dynamic roster row
        const newRow = document.createElement('tr');
        newRow.innerHTML = `
          <td><span class="dot purple"></span>${selectedDept} (Elective)</td>
          <td>${selectedWeeks} Weeks</td>
          <td><span class="pill-badge blue-light">Elective</span></td>
          <td style="text-align: right;">
            <button type="button" class="btn-text btn-remove-row" style="color: #ef4444; font-size: 10px; font-weight: 600;">Remove</button>
          </td>
        `;

        // Attach dynamic deletion listener
        newRow.querySelector('.btn-remove-row').addEventListener('click', function () {
          newRow.remove();
          recalculateTotalWeeks();
        });

        rosterBody.appendChild(newRow);
        recalculateTotalWeeks();
        departmentSelect.value = '';
      });

      function recalculateTotalWeeks() {
        const totalWeeksBadge = document.getElementById('total-weeks-badge');
        let total = 0;
        const rows = rosterBody.querySelectorAll('tr');
        
        rows.forEach(row => {
          const text = row.cells[1] ? row.cells[1].textContent : '';
          const match = text.match(/\d+/);
          if (match) {
            total += parseInt(match[0], 10);
          }
        });

        if (totalWeeksBadge) {
          totalWeeksBadge.textContent = `${total} Weeks Configured`;
        }
      }
    }

    /**
     * 3. Dossier Checkbox Sync & Counter Logic
     */
    function initDocumentVerification() {
      const docCheckboxes = document.querySelectorAll('.doc-check-input');
      const verifiedCounter = document.getElementById('verified-docs-count');

      if (!docCheckboxes.length || !verifiedCounter) return;

      docCheckboxes.forEach(checkbox => {
        checkbox.addEventListener('change', () => {
          const checkedCount = document.querySelectorAll('.doc-check-input:checked').length;
          verifiedCounter.textContent = `${checkedCount}/${docCheckboxes.length} Verified`;

          if (checkedCount === docCheckboxes.length) {
            verifiedCounter.className = 'pill-badge green-light';
          } else {
            verifiedCounter.className = 'pill-badge yellow';
          }
        });
      });
    }

    /**
     * 4. License Authorization Submission Logic
     */
    function initFormSubmission() {
      const authForm = document.getElementById('licensing-induction-form');
      const ethicalCheckbox = document.getElementById('ethical-compliance-check');
      const submitBtn = document.getElementById('btn-authorize-primary');

      if (!authForm || !submitBtn) return;

      authForm.addEventListener('submit', (e) => {
        e.preventDefault();

        if (ethicalCheckbox && !ethicalCheckbox.checked) {
          alert('You must check the compliance verification box before authorizing.');
          return;
        }

        submitBtn.disabled = true;
        submitBtn.innerHTML = 'Processing Order...';

        setTimeout(() => {
          alert('Provisional License PR-BMDC-2025-08492 and Internship Order successfully issued!');
          submitBtn.innerHTML = '✓ Authorized & Issued';
          submitBtn.style.backgroundColor = '#059669';
        }, 1200);
      });
    }
  </script>
