    <script>
        // 1. DOM Initialization
        document.addEventListener('DOMContentLoaded', function() {
            initSidebarNavigation();
        });

        /**
         * Dynamic Sidebar Active State Manager
         */
        function initSidebarNavigation() {
            const navItems = document.querySelectorAll('.nav-menu .nav-item');
            navItems.forEach(item => {
                item.addEventListener('click', function(e) {
                    e.preventDefault();
                    navItems.forEach(nav => nav.classList.remove('active'));
                    this.classList.add('active');
                });
            });
        }

        /**
         * Filter Broadcast Feed Items
         * @param {string} category - 'all', 'emergency', 'advisory', 'info'
         * @param {HTMLElement} element - Button clicked
         */
        function filterFeed(category, element) {
            // Update Active Filter Button
            const buttons = document.querySelectorAll('.filter-bar .filter-btn');
            buttons.forEach(btn => btn.classList.remove('active'));
            element.classList.add('active');

            // Filter Feed Items
            const items = document.querySelectorAll('.feed-item');
            items.forEach(item => {
                const itemType = item.getAttribute('data-type');
                if (category === 'all' || itemType === category) {
                    item.style.display = 'block';
                } else {
                    item.style.display = 'none';
                }
            });
        }

        /**
         * Live Search Broadcast Feed by Keyword or ID
         * @param {string} query
         */
        function handleSearch(query) {
            const searchTerm = query.toLowerCase().trim();
            const items = document.querySelectorAll('.feed-item');

            items.forEach(item => {
                const text = item.textContent.toLowerCase();
                if (text.includes(searchTerm)) {
                    item.style.display = 'block';
                } else {
                    item.style.display = 'none';
                }
            });
        }

        /**
         * Action Handler for Alert Feed Items
         * @param {string} broadcastId
         * @param {string} newStatus
         */
        function updateAlertStatus(broadcastId, newStatus) {
            const statusLabel = document.getElementById(`status-${broadcastId}`);
            if (statusLabel) {
                statusLabel.textContent = newStatus;
                if (newStatus === 'Resolved' || newStatus === 'Completed') {
                    statusLabel.className = 'status-chip success-text';
                } else {
                    statusLabel.className = 'status-chip';
                }
            }
        }

        /**
         * Refresh Animation & Trigger Handler
         */
        function refreshFeed() {
            const icon = document.querySelector('#refreshBtn i');
            if (icon) {
                icon.style.transition = 'transform 0.5s ease';
                icon.style.transform = 'rotate(360deg)';
                setTimeout(() => {
                    icon.style.transform = 'rotate(0deg)';
                }, 500);
            }
            console.log('Broadcast feed synchronized.');
        }

        /**
         * Switch Role View (Admin, Exec, Doc, Surgeon)
         * @param {string} role
         */
        function setAppRole(role) {
            const sidebar = document.getElementById('appSidebar');
            const roleLabel = document.getElementById('userRoleTitle');
            
            if (!sidebar) return;

            sidebar.classList.remove('admin', 'executive', 'doctor', 'surgeon');

            switch (role.toLowerCase()) {
                case 'admin':
                    sidebar.classList.add('admin');
                    roleLabel.textContent = 'Administrator';
                    break;
                case 'executive':
                    sidebar.classList.add('executive');
                    roleLabel.textContent = 'Executive Officer';
                    break;
                case 'doctor':
                    sidebar.classList.add('doctor');
                    roleLabel.textContent = 'Attending Physician';
                    break;
                case 'surgeon':
                    sidebar.classList.add('surgeon');
                    roleLabel.textContent = 'Lead Surgeon';
                    break;
            }
        }

        /**
         * User Session Logout Handler
         */
        function handleLogout() {
            if (confirm('Are you sure you want to log out of Med-Core?')) {
                alert('Session terminated.');
                // window.location.href = '/login.html';
            }
        }
    </script>
